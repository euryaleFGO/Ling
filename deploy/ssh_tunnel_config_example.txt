# SSH 隧道配置示例
# 将此内容添加到 ~/.ssh/config 文件中

# ============================================================
# AI 服务器（ASR + TTS 在同一台服务器）
# ============================================================
Host ai-server
    HostName your-server.com          # 替换为你的服务器地址
    User your-username                # 替换为你的用户名
    Port 22
    
    # ASR 服务器端口映射
    LocalForward 18889 localhost:5002
    
    # TTS 服务器端口映射
    LocalForward 18888 localhost:5001
    
    # 保持连接活跃
    ServerAliveInterval 60
    ServerAliveCountMax 3
    
    # 可选：使用密钥认证
    # IdentityFile ~/.ssh/id_rsa

# ============================================================
# 单独的 ASR 服务器
# ============================================================
Host asr-server
    HostName asr.your-server.com
    User your-username
    Port 22
    LocalForward 18889 localhost:5002
    ServerAliveInterval 60
    ServerAliveCountMax 3

# ============================================================
# 单独的 TTS 服务器
# ============================================================
Host tts-server
    HostName tts.your-server.com
    User your-username
    Port 22
    LocalForward 18888 localhost:5001
    ServerAliveInterval 60
    ServerAliveCountMax 3

# ============================================================
# 使用方法
# ============================================================
# 1. 复制此文件内容到 ~/.ssh/config
# 2. 修改 HostName 和 User 为实际值
# 3. 连接隧道：
#    ssh -fN ai-server      # 后台运行
#    ssh ai-server          # 前台运行（可以看到日志）
#
# 4. 验证连接：
#    curl http://localhost:18889/health  # ASR
#    curl http://localhost:18888/health  # TTS
#
# 5. 关闭隧道：
#    ps aux | grep "ssh.*ai-server"
#    kill <PID>

# ============================================================
# Windows 用户注意事项
# ============================================================
# 1. 配置文件位置：C:\Users\YourName\.ssh\config
# 2. 使用 PowerShell 或 Git Bash 执行 ssh 命令
# 3. 查看端口占用：netstat -an | findstr 18889
# 4. 杀死进程：taskkill /F /PID <PID>
